// Miner Wars Discord Bot
const{Client,GatewayIntentBits,EmbedBuilder}=require('discord.js');
const fetch=require('node-fetch');
const CONFIG={BOT_TOKEN:'YOUR_TOKEN',CHANNEL:'YOUR_CHANNEL_ID'};
const client=new Client({intents:[GatewayIntentBits.Guilds,GatewayIntentBits.GuildMessages,GatewayIntentBits.MessageContent]});
let lastBlock=0,boost=null,ready=new Set();
const getBlock=async()=>{try{return(await(await fetch('https://mempool.space/api/v1/blocks')).json())[0]}catch{return null}};
const getPrice=async()=>{try{return await(await fetch('https://api.coingecko.com/api/v3/simple/price?ids=gomining-token,bitcoin&vs_currencies=usd&include_24hr_change=true')).json()}catch{return null}};
async function check(){const b=await getBlock();if(!b)return;const ch=client.channels.cache.get(CONFIG.CHANNEL);if(b.height>lastBlock&&lastBlock>0&&ch)ch.send({content:'@here',embeds:[new EmbedBuilder().setColor(0x10B981).setTitle('⛏️ NEW BLOCK!').setDescription(`Block ${b.height} - New round!`)]});lastBlock=b.height}
client.on('messageCreate',async m=>{if(m.author.bot||!m.content.startsWith('!'))return;const[c,...a]=m.content.slice(1).split(' ');
if(c==='block'){const b=await getBlock();if(!b)return m.reply('❌');const t=((Date.now()-b.timestamp*1000)/60000).toFixed(1);m.reply({embeds:[new EmbedBuilder().setColor(0xFBBF24).setTitle('⏱️ Block').addFields({name:'Height',value:b.height.toString(),inline:true},{name:'Time',value:t+' min',inline:true},{name:'Status',value:t<5?'🟢':t<10?'🟡':'🔴',inline:true})]})}
if(c==='price'){const p=await getPrice();if(!p)return m.reply('❌');const g=p['gomining-token'];m.reply(`💰 GMT: $${g.usd.toFixed(4)} (${g.usd_24h_change>=0?'+':''}${g.usd_24h_change.toFixed(2)}%) | BTC: $${p.bitcoin.usd.toLocaleString()}`)}
if(c==='spell'){const b=await getBlock();if(!b)return;const t=(Date.now()-b.timestamp*1000)/60000;m.reply(t<3?'🚀 ROCKET - Max duration!':t<6?'🌀 ECHO - Good for long':t<10?'⚡ INSTANT - Safe':'🎯 FOCUS - Risky!')}
if(c==='boost'){boost={m:a[0]||'x8'};ready.clear();ready.add(m.author.id);m.channel.send({content:'@here',embeds:[new EmbedBuilder().setColor(0x8B5CF6).setTitle('🚀 BOOST '+boost.m).setDescription('Type !ready to join')]})}
if(c==='ready'){if(!boost)return m.reply('No session');ready.add(m.author.id);m.reply(`✅ ${ready.size} ready`)}
if(c==='go'){if(!boost)return;m.channel.send(`@here 🔥 BOOST NOW! ${ready.size} GO!`);boost=null}
if(c==='help')m.reply('!block !price !spell !boost !ready !go')});
client.once('ready',()=>{console.log('⛏️ Online');setInterval(check,30000)});
client.login(CONFIG.BOT_TOKEN);
